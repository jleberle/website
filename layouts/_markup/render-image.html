{{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $fullSrc := "" -}} {{- $srcset := "" -}} {{- $sizes := "" -}} {{- $imgWidth := "" -}} {{- $imgHeight := "" -}} {{- $processable := slice "jpg" "jpeg" "png" "tif" "bmp" "gif" "webp" "avif" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path }} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- if in $processable .MediaType.SubType -}} {{- $r := partial "responsive_img.html" (dict "resource" .) -}} {{- $fullSrc = $r.full -}} {{- $src = $r.src -}} {{- $srcset = $r.srcset -}} {{- $sizes = $r.sizes -}} {{- $imgWidth = $r.width -}} {{- $imgHeight = $r.height -}} {{- else -}} {{- /* Non-raster (e.g. SVG): serve fingerprinted as-is, no resizing. */ -}} {{- $full := . | fingerprint -}} {{- $fullSrc = $full.RelPermalink -}} {{- $src = $full.RelPermalink -}} {{- end -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Defaults first, then author attributes win: a leading/hero image marked `{loading="eager" fetchpriority="high"}` overrides the lazy default so the LCP image isn't lazy-loaded. Computed src/srcset/width/height are merged on top below, so those can't be clobbered from content. */ -}} {{- $attributes := merge (dict "alt" .Text "src" $src "title" (.Title | transform.HTMLEscape) "loading" "lazy" "decoding" "async") .Attributes -}} {{- if $fullSrc -}} {{- $attributes = merge $attributes (dict "data-lightbox-src" $fullSrc) -}} {{- end -}} {{- if $srcset -}} {{- $attributes = merge $attributes (dict "srcset" $srcset "sizes" $sizes) -}} {{- end -}} {{- if $imgWidth -}} {{- $attributes = merge $attributes (dict "width" $imgWidth "height" $imgHeight) -}} {{- end -}} {{- /**/ -}}
