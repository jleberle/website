{{- $id := .Get 0 -}} {{- $thumbnail := "" -}} {{- $tw := 0 -}} {{- $th := 0 -}} {{- $maxres := printf "https://img.youtube.com/vi/%s/maxresdefault.jpg" $id -}} {{- $hqdefault := printf "https://img.youtube.com/vi/%s/hqdefault.jpg" $id -}} {{- /* Re-encode the remote JPG thumbnail to AVIF (format-only, native width — no upscaling) and copy under /images/ so the immutable Cache-Control rule in static/_headers applies (a bare GetRemote would publish at the site root). */ -}} {{- with resources.GetRemote $maxres -}} {{- $t := .Resize (printf "%dx avif" .Width) | resources.Copy (printf "images/yt-%s.avif" $id) | fingerprint -}} {{- $thumbnail = $t.RelPermalink -}} {{- $tw = $t.Width -}} {{- $th = $t.Height -}} {{- else -}} {{- with resources.GetRemote $hqdefault -}} {{- $t := .Resize (printf "%dx avif" .Width) | resources.Copy (printf "images/yt-%s.avif" $id) | fingerprint -}} {{- $thumbnail = $t.RelPermalink -}} {{- $tw = $t.Width -}} {{- $th = $t.Height -}} {{- end -}} {{- end -}}
