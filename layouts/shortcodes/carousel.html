{{- /* Resolve params to page resources up front so slides and dots always pair up. */ -}} {{- $images := slice -}} {{- range .Params -}} {{- with $.Page.Resources.GetMatch . -}}{{- $images = $images | append . -}}{{- end -}} {{- end -}}
{{- range $i, $img := $images }} {{- /* Native AVIF srcset: smaller widths for phones, original as the largest candidate. */}} {{- $srcsetParts := slice }} {{- if gt $img.Width 400 }}{{ $r := $img.Resize "400x avif" | fingerprint }}{{ $srcsetParts = $srcsetParts | append (printf "%s 400w" $r.RelPermalink) }}{{ end }} {{- if gt $img.Width 800 }}{{ $r := $img.Resize "800x avif" | fingerprint }}{{ $srcsetParts = $srcsetParts | append (printf "%s 800w" $r.RelPermalink) }}{{ end }} {{- $processed := $img | fingerprint }} {{- $srcsetParts = $srcsetParts | append (printf "%s %dw" $processed.RelPermalink $img.Width) }} {{- end }}
