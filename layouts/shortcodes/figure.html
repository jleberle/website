{{- /* Responsive override of PaperMod's figure shortcode: when src resolves to a processable page/asset resource, emit a native AVIF srcset + dimensions (mirrors layouts/_markup/render-image.html); otherwise falls back to a plain exactly like the theme. Keeps the #center fragment for the existing CSS centering rule and all of the theme's align/link/caption features. */ -}} {{- $src := .Get "src" -}} {{- $align := eq (.Get "align") "center" -}} {{- $img := "" -}} {{- with $src }}{{ $img = or ($.Page.Resources.GetMatch .) (resources.Get .) }}{{ end -}} {{- $srcset := "" -}} {{- $imgW := .Get "width" -}} {{- $imgH := .Get "height" -}} {{- $finalSrc := $src -}} {{- $processable := slice "jpg" "jpeg" "png" "tif" "bmp" "gif" "webp" "avif" -}} {{- if and $img (in $processable $img.MediaType.SubType) -}} {{- $full := $img | fingerprint -}} {{- $finalSrc = $full.RelPermalink -}} {{- $imgW = string $img.Width -}} {{- $imgH = string $img.Height -}} {{- $parts := slice -}} {{- if gt $img.Width 1200 }}{{ $r := $img.Resize "1200x avif" | fingerprint }}{{ $parts = $parts | append (printf "%s 1200w" $r.RelPermalink) }}{{ end -}} {{- if gt $img.Width 800 }}{{ $r := $img.Resize "800x avif" | fingerprint }}{{ $parts = $parts | append (printf "%s 800w" $r.RelPermalink) }}{{ end -}} {{- if gt $img.Width 400 }}{{ $r := $img.Resize "400x avif" | fingerprint }}{{ $parts = $parts | append (printf "%s 400w" $r.RelPermalink) }}{{ end -}} {{- $parts = $parts | append (printf "%s %dw" $full.RelPermalink $img.Width) -}} {{- if gt (len $parts) 1 }}{{ $srcset = delimit $parts ", " }}{{ end -}} {{- end -}} {{- if $align }}{{ $finalSrc = printf "%s#center" $finalSrc }}{{ end -}} {{- if .Get "link" -}} {{- end }} {{- if .Get "link" }}{{ end -}} {{- if or (or (.Get "title") (.Get "caption")) (.Get "attr") -}} {{- end }}
