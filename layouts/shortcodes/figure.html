{{- /* Responsive override of PaperMod's figure shortcode: when src resolves to a processable page/asset resource, emit a native AVIF srcset + dimensions (mirrors layouts/_markup/render-image.html); otherwise falls back to a plain exactly like the theme. Keeps the #center fragment for the existing CSS centering rule and all of the theme's align/link/caption features. */ -}} {{- $src := .Get "src" -}} {{- $align := eq (.Get "align") "center" -}} {{- $img := "" -}} {{- with $src }}{{ $img = or ($.Page.Resources.GetMatch .) (resources.Get .) }}{{ end -}} {{- $srcset := "" -}} {{- $sizes := "" -}} {{- $fullSrc := "" -}} {{- $imgW := .Get "width" -}} {{- $imgH := .Get "height" -}} {{- $finalSrc := $src -}} {{- $processable := slice "jpg" "jpeg" "png" "tif" "bmp" "gif" "webp" "avif" -}} {{- if and $img (in $processable $img.MediaType.SubType) -}} {{- $r := partial "responsive-img.html" (dict "resource" $img) -}} {{- $finalSrc = $r.src -}} {{- $srcset = $r.srcset -}} {{- $sizes = $r.sizes -}} {{- $fullSrc = $r.full -}} {{- $imgW = $r.width -}} {{- $imgH = $r.height -}} {{- else if $img -}} {{- /* Non-raster (e.g. SVG): no resizing, but still invoke RelPermalink so the file is published under build.publishResources=false. */ -}} {{- $full := $img | fingerprint -}} {{- $finalSrc = $full.RelPermalink -}} {{- end -}} {{- if $align }}{{ $finalSrc = printf "%s#center" $finalSrc }}{{ end -}} {{- /* alt is always emitted but never copied from the caption: alt describes what the image shows, the caption is visible attribution/commentary, and duplicating one into the other makes screen readers announce the same text twice. A captioned figure with alt="" is conforming HTML — the figcaption is the programmatic description. */ -}} {{- if not (or (.Get "alt") (.Get "caption")) -}} {{- warnf "figure missing alt and caption: %s (%s)" .Page.RelPermalink $src -}} {{- end -}} {{- if .Get "link" -}} {{- end }} {{- if .Get "link" }}{{ end -}} {{- if or (or (.Get "title") (.Get "caption")) (.Get "attr") -}} {{- end }}
