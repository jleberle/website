{{- /* Shared per-item syndication content for the home RSS and JSON feeds. Takes a page as context, returns dict(content, cover): - content: page HTML with responsive/lightbox attributes stripped and root-relative/fragment URLs absolutized (feed readers have no page base to resolve against), led by the frontmatter cover image unless the body already embeds it. - cover: the resolved cover permalink, or "" if none. Both feeds need the same transform, so it lives in one place. */ -}} {{- $page := . -}} {{- $base := site.BaseURL | strings.TrimSuffix "/" -}} {{- $cover := "" -}} {{- with $page.Params.cover.image -}} {{- with $page.Resources.GetMatch . -}}{{ $cover = (. | resources.Fingerprint).Permalink }}{{- else -}}{{ $cover = print $page.Permalink . }}{{- end -}} {{- end -}} {{- $c := $page.Content | replaceRE `\s+(srcset|sizes|data-lightbox-src|loading|decoding)="[^"]*"` "" | replaceRE `href="#([^"]*)"` (printf `href="%s#$1"` $page.Permalink) | replaceRE `(src|href)="(/[^"]*)"` (printf `$1="%s$2"` $base) -}} {{- $coverImg := "" -}} {{- with $cover -}} {{- $stem := strings.TrimSuffix (path.Ext $page.Params.cover.image) (path.Base $page.Params.cover.image) -}} {{- if not (or (strings.Contains $c (printf "/%s." $stem)) (strings.Contains $c (printf "/%s_" $stem))) -}} {{- $alt := $page.Params.cover.alt | default $page.Title -}} {{- $coverImg = printf "\n" $cover $alt "max-width:100%;height:auto;" -}} {{- end -}} {{- end -}} {{- /* Feed readers don't load the site stylesheet, so the review's source citation (normally .post-context, styled via CSS) is rebuilt here with inline styles and prepended to the body, mirroring single.html's placement of post_context.html above the content. It's inserted after the cover image (rather than above it) so the photo leads the item. */ -}} {{- if eq $page.Section "reviews" -}} {{- with $page.GetTerms "sources" -}} {{- $source := index . 0 -}} {{- $p := $source.Params -}} {{- $citeAuthor := "" -}} {{- with $p.author }}{{ $citeAuthor = printf "%s, " . }}{{ end -}} {{- $citeYear := "" -}} {{- with $p.published_year }}{{ $citeYear = printf "%v" . }}{{ end -}} {{- $citeRecord := "" -}} {{- if $p.publisher -}} {{- $citeRecord = printf " (%s%s)" $p.publisher (cond (ne $citeYear "") (printf ", %s" $citeYear) "") -}} {{- else if $citeYear -}} {{- $citeRecord = printf " (%s)" $citeYear -}} {{- end -}} {{- $kicker := printf "Reviewed %s" ($p.type | default "book" | lower) -}} {{- $citation := printf "

%s

\n

%s%s%s

" "font-size:0.85em;text-transform:uppercase;letter-spacing:.04em;color:#666;margin:0 0 .25em" $kicker "margin:0 0 1em" $citeAuthor $source.Permalink $source.Title $citeRecord -}} {{- $c = printf "%s\n%s" $citation $c -}} {{- end -}} {{- end -}} {{- $c = printf "%s%s" $coverImg $c -}} {{- $email := site.Params.email -}} {{- $mailto := printf "mailto:%s?subject=%s" $email (urlquery (printf "Re: %s" $page.Title)) -}} {{- $license := site.Copyright | markdownify -}} {{- $c = printf "%s\n

Reply by email · © %d %s. Licensed under %s.

\n

Originally published at %s.

" $c $mailto $page.Date.Year site.Params.author $license $page.Permalink $page.Permalink -}} {{- return (dict "content" $c "cover" $cover) -}}
