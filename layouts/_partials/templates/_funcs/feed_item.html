{{- /* Shared per-item syndication content for the home RSS and JSON feeds. Takes a page as context, returns dict(content, cover, markdown): - content: page HTML with responsive/lightbox attributes stripped and root-relative/fragment URLs absolutized (feed readers have no page base to resolve against), led by the frontmatter cover image unless the body already embeds it. - cover: the resolved cover permalink, or "" if none. - markdown: the same item as Markdown (RSS source:markdown / JSON Feed content_text) with the same URL absolutization and cover/citation/footer treatment as content, or "" when the page uses a shortcode — .RawContent is pre-expansion, so a shortcode would otherwise leak into the feed as literal shortcode syntax text. See reading.rss.xml's old source:markdown comment for why that extension was dropped from the reading feed; this is a separate, direct-from-Markdown-source case, not that one. Both feeds need the same transform, so it lives in one place. */ -}} {{- $page := . -}} {{- $base := site.BaseURL | strings.TrimSuffix "/" -}} {{- $cover := "" -}} {{- with $page.Params.cover.image -}} {{- with $page.Resources.GetMatch . -}}{{ $cover = (. | resources.Fingerprint).Permalink }}{{- else -}}{{ $cover = print $page.Permalink . }}{{- end -}} {{- end -}} {{- $c := $page.Content | replaceRE `\s+(srcset|sizes|data-lightbox-src|loading|decoding)="[^"]*"` "" | replaceRE `href="#([^"]*)"` (printf `href="%s#$1"` $page.Permalink) | replaceRE `(src|href)="(/[^"]*)"` (printf `$1="%s$2"` $base) -}} {{- $coverImg := "" -}} {{- $coverStem := "" -}} {{- with $cover -}} {{- $coverStem = strings.TrimSuffix (path.Ext $page.Params.cover.image) (path.Base $page.Params.cover.image) -}} {{- if not (or (strings.Contains $c (printf "/%s." $coverStem)) (strings.Contains $c (printf "/%s_" $coverStem))) -}} {{- $alt := $page.Params.cover.alt | default $page.Title -}} {{- $coverImg = printf "\n" $cover $alt "max-width:100%;height:auto;" -}} {{- end -}} {{- end -}} {{- /* Feed readers don't load the site stylesheet, so the review's source citation (normally .post-context, styled via CSS) is rebuilt here with inline styles and prepended to the body, mirroring single.html's placement of post_context.html above the content. It's inserted after the cover image (rather than above it) so the photo leads the item. */ -}} {{- $citeAuthor := "" -}} {{- $citeYear := "" -}} {{- $citeRecord := "" -}} {{- $kicker := "" -}} {{- $citeSource := false -}} {{- if eq $page.Section "reviews" -}} {{- with $page.GetTerms "sources" -}} {{- $citeSource = index . 0 -}} {{- $p := $citeSource.Params -}} {{- with $p.author }}{{ $citeAuthor = printf "%s, " . }}{{ end -}} {{- with $p.published_year }}{{ $citeYear = printf "%v" . }}{{ end -}} {{- if $p.publisher -}} {{- $citeRecord = printf " (%s%s)" $p.publisher (cond (ne $citeYear "") (printf ", %s" $citeYear) "") -}} {{- else if $citeYear -}} {{- $citeRecord = printf " (%s)" $citeYear -}} {{- end -}} {{- $kicker = printf "Reviewed %s" ($p.type | default "book" | lower) -}} {{- $citation := printf "

%s

\n

%s%s%s

" "font-size:0.85em;text-transform:uppercase;letter-spacing:.04em;color:#666;margin:0 0 .25em" $kicker "margin:0 0 1em" $citeAuthor $citeSource.Permalink $citeSource.Title $citeRecord -}} {{- $c = printf "%s\n%s" $citation $c -}} {{- end -}} {{- end -}} {{- $c = printf "%s%s" $coverImg $c -}} {{- $email := site.Params.email -}} {{- $mailto := printf "mailto:%s?subject=%s" $email (urlquery (printf "Re: %s" $page.Title)) -}} {{- $license := site.Copyright | markdownify -}} {{- $c = printf "%s\n

Reply by email · © %d %s. Licensed under %s.

\n

Originally published at %s.

" $c $mailto $page.Date.Year site.Params.author $license $page.Permalink $page.Permalink -}} {{- $md := "" -}} {{- if not (or (strings.Contains $page.RawContent "{{<") (strings.Contains $page.RawContent "{{%")) -}} {{- /* Same three URL cases as $c above, but for Markdown link/image syntax "](...)" instead of HTML attributes. Ordered so root-relative and anchor paths are absolutized (and so start with "https://", i.e. contain a colon) before the bundle-relative pass runs — that pass is scoped to paths with no colon at all, which is what distinguishes a plain page-bundle filename (cover.avif) from anything already carrying a scheme (https://, mailto:, doi:) with no lookahead needed, since RE2 doesn't support it. */ -}} {{- $m := $page.RawContent | replaceRE `(?i)\n?\n?` "\n" | replaceRE `\]\(#([^)]*)\)` (printf `](%s#$1)` $page.Permalink) | replaceRE `\]\(/([^)]*)\)` (printf `](%s/$1)` $base) | replaceRE `\]\(([a-zA-Z0-9_][^):]*)\)` (printf `](%s$1)` $page.Permalink) -}} {{- $mdCoverImg := "" -}} {{- with $cover -}} {{- if not (or (strings.Contains $m (printf "/%s." $coverStem)) (strings.Contains $m (printf "/%s_" $coverStem))) -}} {{- $alt := $page.Params.cover.alt | default $page.Title -}} {{- $mdCoverImg = printf "![%s](%s)\n\n" $alt $cover -}} {{- end -}} {{- end -}} {{- if $citeSource -}} {{- $mdCitation := printf "*%s*\n\n%s[*%s*](%s)%s\n\n" $kicker $citeAuthor $citeSource.Title $citeSource.Permalink $citeRecord -}} {{- $m = printf "%s%s" $mdCitation $m -}} {{- end -}} {{- $m = printf "%s%s" $mdCoverImg $m -}} {{- $md = printf "%s\n\n[Reply by email](%s) · © %d %s. Licensed under %s.\n\nOriginally published at [%s](%s)." $m $mailto $page.Date.Year site.Params.author site.Copyright $page.Permalink $page.Permalink -}} {{- end -}} {{- return (dict "content" $c "cover" $cover "markdown" $md) -}}
