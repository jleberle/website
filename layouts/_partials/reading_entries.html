{{- $entries := slice -}} {{- range $collectionName, $collection := hugo.Data.reading }} {{- $defaultType := lower (replaceRE "s$" "" $collectionName) -}} {{- range $key, $source := $collection }} {{- if not (hasPrefix $key "_") }} {{- $slug := $key -}} {{- $type := lower ($source.type | default $defaultType) -}} {{- $typeLabel := humanize $type -}} {{- $status := lower ($source.status | default "read") -}} {{- $publishedYear := int ($source.published_year | default 0) -}} {{- $readYear := int ($source.read_year | default 0) -}} {{- $sortRead := cond (gt $readYear 0) $readYear 0 -}} {{- $sortPublished := cond (gt $publishedYear 0) $publishedYear 0 -}} {{- $finishedSort := $source.finished | default "0000-00-00" -}} {{- $sortReadKey := printf "%04d-%s-%s-%s" $sortRead $finishedSort $type $slug -}} {{- $sortPublishedKey := printf "%04d-%s-%s-%s" $sortPublished $finishedSort $type $slug -}} {{- $anchorID := printf "%s-%s" $type $slug -}} {{- $entries = $entries | append (dict "slug" $slug "anchor_id" $anchorID "collection" $collectionName "type" $type "type_label" $typeLabel "title" ($source.title | default $slug) "author" ($source.author | default "") "cite_key" ($source.cite_key | default "") "status" $status "published_year" $publishedYear "read_year" $readYear "publisher" ($source.publisher | default "") "container_title" ($source.container_title | default "") "volume" ($source.volume | default "") "issue" ($source.issue | default "") "pages" ($source.pages | default "") "isbn" ($source.isbn | default "") "doi" ($source.doi | default "") "url" ($source.url | default "") "format" ($source.format | default "") "started" ($source.started | default "") "started_announced" ($source.started_announced | default "") "finished" ($source.finished | default "") "finished_announced" ($source.finished_announced | default "") "notes" ($source.notes | default "") "related_posts" ($source.related_posts | default (slice)) "sort_read" $sortRead "sort_read_key" $sortReadKey "sort_published" $sortPublished "sort_published_key" $sortPublishedKey ) -}} {{- end }} {{- end }} {{- end }} {{- return $entries -}}
